CREATE DATABASE carona_universitaria DEFAULT CHARACTER SET = 'utf8mb4';

USE carona_universitaria;

CREATE TABLE estudantes (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(255) NOT NULL,
    telefone VARCHAR(11) NOT NULL,
    cpf VARCHAR(11) NOT NULL UNIQUE,
    rgm VARCHAR(10) NOT NULL UNIQUE,
    email VARCHAR(255) NOT NULL UNIQUE,
    senha VARCHAR(255) NOT NULL,
    instituicao VARCHAR(255) NOT NULL,
    curso VARCHAR(255) NOT NULL,
    periodo VARCHAR(255) NOT NULL,
    verificado BOOLEAN NOT NULL DEFAULT FALSE,
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS caronas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    motorista_id INT NOT NULL,
    origem VARCHAR(100),
    destino VARCHAR(100),
    horario VARCHAR(20),
    vagas INT
);

CREATE TABLE IF NOT EXISTS avaliacoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    avaliador_id INT NOT NULL,
    avaliado_id INT NOT NULL,
    carona_id INT NOT NULL,
    nota INT NOT NULL CHECK (nota BETWEEN 1 AND 5),
    comentario TEXT,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (avaliador_id) REFERENCES estudantes (id) ON DELETE CASCADE,
    FOREIGN KEY (avaliado_id) REFERENCES estudantes (id) ON DELETE CASCADE,
    FOREIGN KEY (carona_id) REFERENCES caronas (id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS caronas_participantes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    carona_id INT NOT NULL,
    estudante_id INT NOT NULL,
    valor DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    confirmado_em DATETIME DEFAULT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'pendente',
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (carona_id) REFERENCES caronas (id) ON DELETE CASCADE,
    FOREIGN KEY (estudante_id) REFERENCES estudantes (id) ON DELETE CASCADE
);

-- SELECT: caronas_participantes

SELECT
    c.id AS carona_no,
    m.nome AS motorista,
    c.origem AS partida,
    p.nome AS passageiro,
    cp.status AS status_passageiro
FROM
    caronas c
    JOIN estudantes m ON c.motorista_id = m.id
    JOIN caronas_participantes cp ON cp.carona_id = c.id
    JOIN estudantes p ON cp.estudante_id = p.id;
